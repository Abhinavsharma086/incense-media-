model User {
  id                 String   @id @default(uuid())
  firstName          String
  lastName           String
  email              String   @unique
  password           String
  phone              String?
  isActive           Boolean  @default(true)
  isSuspended        Boolean  @default(false)
  twoFactorEnabled   Boolean  @default(false)
  twoFactorSecret    String?
  forcePasswordReset Boolean  @default(false)
  lastLoginAt        DateTime?
  lastLoginIp        String?
  roleId             String
  role               Role     @relation(fields: [roleId], references: [id])
  leads              Lead[]   @relation("LeadOwner")
  deals              Deal[]   @relation("DealOwner")
  campaigns          MediaCampaign[] @relation("CampaignOwner")
  
  // Unit & Lead Assignment Relations
  unitEmployees      UnitEmployee[]
  assignmentLogsFrom AssignmentLog[] @relation("AssignmentLogFrom")
  assignmentLogsTo   AssignmentLog[] @relation("AssignmentLogTo")
  assignmentLogsBy   AssignmentLog[] @relation("AssignmentLogBy")

  // Super Admin & Granular RBAC Relations
  userPermissions    UserPermission[]
  adminPermissions   AdminPermission[]
  loginHistories     LoginHistory[]
  sessions           UserSession[]
  activityLogs       ActivityLog[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}
